' ==========================================
' ENGINE & CORE INTERFACES
' ==========================================

class GameEngine {
    - width : int
    - height : int
    - title : str
    - target_fps : int
    - screen : pygame.Surface
    - clock : pygame.time.Clock
    - is_running : bool
    - current_state : GameState
    - settings_manager : SettingsManager
    + initialize() : void
    + change_state(new_state: GameState) : void
    + handle_events() : void
    + update(delta_time: float) : void
    + render() : void
    + run() : void
    + quit() : void
}

abstract class GameState {
    # engine : GameEngine
    + __init__(engine: GameEngine)
    + {abstract} handle_input(event: pygame.event.Event) : void
    + {abstract} update(delta_time: float) : void
    + {abstract} draw(surface: pygame.Surface) : void
}

' ==========================================
' CONFIGURATION & PERSISTENCE
' ==========================================

class SettingsManager {
    - file_path : str
    - volume : float
    - key_bindings : Dict[str, int]
    - das_delay : float
    - arr_rate : float
    + load_settings() : void
    + save_settings() : void
    + get_binding(action: str) : int
    + set_binding(action: str, key: int) : void
    + get_volume() : float
    + set_volume(volume: float) : void
}

interface IHighScoreRepository {
    + load_high_score() : int
    + save_high_score(score: int) : void
}

class FileHighScoreRepository implements IHighScoreRepository {
    - file_path : str
    + load_high_score() : int

}

class PlayState {

    + pause_game() : void

class Board {
    - width : int = 10
    - height : int = 20
    - grid : List[List[Optional[Tuple[int, int, int]]]]

    + is_line_full(row: int) : bool
    + clear_line(row: int) : void
    + move_rows_down(from_row: int) : void
    + reset() : void
}

' ==========================================
' RELATIONSHIPS & INHERITANCE
' ==========================================

GameEngine "1" *-- "1" GameState : current_state
GameEngine "1" *-- "1" SettingsManager : owns settings
GameState "1" o-- "1" GameEngine : engine

' Explicit GameState Inheritance Tree
GameState <|-- TitleState
GameState <|-- PlayState
GameState <|-- PauseState
GameState <|-- SettingsState
GameState <|-- GameOverState

' Shared Settings References
SettingsState "1" o-- "1" SettingsManager : modifies & saves
InputHandler "1" o-- "1" SettingsManager : reads keybindings

' PlayState Compositions & Aggregations
PlayState "1" *-- "1" IInputHandler
PlayState "1" *-- "1" IScoreManager
PlayState "1" *-- "1" IChallengeManager
PlayState "1" *-- "1" Board
PlayState "1" *-- "1" BoardRenderer
PlayState "1" o-- "1..3" Piece

' Renderer Interactions
BoardRenderer "1" ..> "1" Board : renders
BoardRenderer "1" ..> "1" Piece : renders

' Domain Interactions
GuidelineScoreManager "1" o-- "1" IHighScoreRepository : repo

' Explicit Ownership of Suspended State
PauseState "1" *-- "1" PlayState : suspended_play_state
